# Formats de fichiers pris en charge
    .pdf .txt .md .csv .tsv .log  (aucune dépendance optionnelle)
    .docx  (nécessite requirements-docx.txt)

# Usage (ligne de commande)
    # Mode découverte : regrouper des documents non triés par similarité (sans exemples au préalable)
    python classeur_documents.py discover --input ./mes_documents --output ./classified

    # Mode entraînement : apprendre un modèle à partir de documents déjà triés (1 sous-dossier = 1 catégorie)
    python classeur_documents.py train --input ./mes_categories --model model.pkl

    # Mode classification : appliquer un modèle déjà entraîné à de nouveaux documents
    python classeur_documents.py classify --input ./nouveaux_fichiers --model model.pkl --output ./classified

# Usage (interface graphique, tout-en-un)
    python classeur_documents_gui.py

    # Onglet "Entraînement" (aucun tri manuel requis)
    #   1) Choisir un dossier de documents en vrac (les catégories sont détectées automatiquement)
    #   2) Choisir le moteur (TF-IDF ou embeddings — explication affichée dans l'onglet)
    #   3) Optionnel : choisir un modèle existant à améliorer avec ces nouvelles données
    #   4) Choisir le nom du modèle à créer (.pkl), puis "Entraîner le modèle"

    # Onglet "Classification"
    #   1) Charger un modèle (.pkl) déjà entraîné (catégories détectées et fichiers du dataset visibles juste en dessous)
    #   2) "Ajouter des fichiers.../Ajouter un dossier..."
    #   3) Corriger d'un double-clic (ou "Attribuer une catégorie..." pour une sélection multiple) les catégories mal proposées
    #   4) "Valider et classer" pour ranger les fichiers dans le dossier de sortie

    # Onglet "Automatisation" (plusieurs automatisations en parallèle)
    #   1) "Ajouter..." : dossier à surveiller, modèle à appliquer, dossier de sortie,
    #      intervalle (toutes les X minutes/heures/jours)
    #   2) L'automatisation tourne en tâche de fond dès l'ajout, et reprend
    #      automatiquement au prochain lancement de l'application
